cmake_minimum_required(VERSION 3.25)
project(ortools_backend_test LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(ortools CONFIG REQUIRED)
add_executable(ortools_backend_test solver_boundary.cpp)
get_target_property(ORTOOLS_LIBRARY_TYPE ortools::ortools TYPE)
if(ORTOOLS_LIBRARY_TYPE STREQUAL "STATIC_LIBRARY")
  target_link_libraries(ortools_backend_test PRIVATE "$<LINK_LIBRARY:WHOLE_ARCHIVE,ortools::ortools>")
else()
  target_link_libraries(ortools_backend_test PRIVATE ortools::ortools)
endif()
if(ORTOOLS_BUNDLE_DIR)
  set_target_properties(ortools_backend_test PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${ORTOOLS_BUNDLE_DIR}")
endif()
if(APPLE)
  set_target_properties(ortools_backend_test PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "@loader_path/lib")
else()
  set_target_properties(ortools_backend_test PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN/lib")
endif()
