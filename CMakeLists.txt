cmake_minimum_required(VERSION 3.20)
project(ortools_duckdb)
set(CMAKE_CXX_STANDARD 20)
find_package(ortools CONFIG REQUIRED)
include_directories(src/include)
set(ORTOOLS_SOURCES src/ortools_extension.cpp src/solver.cpp)
build_static_extension(ortools ${ORTOOLS_SOURCES})
build_loadable_extension(ortools "" ${ORTOOLS_SOURCES})
foreach(target ortools_extension ortools_loadable_extension)
  target_link_libraries(${target} ortools::ortools)
  if(ORTOOLS_DUCKDB_1_4)
    target_compile_definitions(${target} PRIVATE ORTOOLS_DUCKDB_1_4=1)
  endif()
endforeach()
if(ORTOOLS_DUCKDB_STATIC_DIR)
  target_link_libraries(ortools_loadable_extension "${ORTOOLS_DUCKDB_STATIC_DIR}/libduckdb_static.a")
  if(ORTOOLS_DUCKDB_1_4)
    file(GLOB dependencies "${ORTOOLS_DUCKDB_STATIC_DIR}/*.a")
    list(REMOVE_ITEM dependencies "${ORTOOLS_DUCKDB_STATIC_DIR}/libduckdb_static.a")
    target_link_libraries(ortools_loadable_extension ${dependencies})
  else()
    target_link_libraries(ortools_loadable_extension dummy_static_extension_loader)
  endif()
endif()
if(APPLE)
  set_target_properties(ortools_loadable_extension PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "@loader_path/lib")
elseif(UNIX)
  set_target_properties(ortools_loadable_extension PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN/lib" CXX_VISIBILITY_PRESET hidden)
  target_link_options(ortools_loadable_extension PRIVATE -Wl,--gc-sections -Wl,--exclude-libs,ALL)
endif()
install(TARGETS ortools_extension EXPORT "${DUCKDB_EXPORT_SET}" LIBRARY DESTINATION "${INSTALL_LIB_DIR}" ARCHIVE DESTINATION "${INSTALL_LIB_DIR}")
