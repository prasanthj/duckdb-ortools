cmake_minimum_required(VERSION 3.25)
project(ortools_duckdb)
set(CMAKE_CXX_STANDARD 20)
if(NOT ortools_DIR OR ortools_DIR STREQUAL "${CMAKE_CURRENT_SOURCE_DIR}/vendor/ortools-minimal-static/lib/cmake/ortools")
  include(cmake/community_dependencies.cmake)
endif()
find_package(ortools CONFIG REQUIRED)
get_target_property(ORTOOLS_LIBRARY_TYPE ortools::ortools TYPE)
set(ORTOOLS_LINK_TARGET ortools::ortools)
if(ORTOOLS_LIBRARY_TYPE STREQUAL "STATIC_LIBRARY")
  # Solver factory registrations have no direct references and must not be discarded.
  set(ORTOOLS_LINK_TARGET "$<LINK_LIBRARY:WHOLE_ARCHIVE,ortools::ortools>")
endif()
get_filename_component(ORTOOLS_PREFIX "${ortools_DIR}/../../.." ABSOLUTE)
set(ORTOOLS_NOTICES "")
file(GLOB_RECURSE ORTOOLS_DEPENDENCY_NOTICES "${ORTOOLS_PREFIX}/share/licenses/*")
list(APPEND ORTOOLS_DEPENDENCY_NOTICES "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE"
  "${CMAKE_CURRENT_SOURCE_DIR}/NOTICE" "${CMAKE_CURRENT_SOURCE_DIR}/third_party/LICENSE.duckdb"
  "${CMAKE_CURRENT_SOURCE_DIR}/src/include/LICENSE.nlohmann-json")
foreach(notice IN LISTS ORTOOLS_DEPENDENCY_NOTICES)
  if(NOT notice MATCHES "\\.patch$")
    file(READ "${notice}" notice_text)
    file(RELATIVE_PATH notice_name "${ORTOOLS_PREFIX}/share/licenses" "${notice}")
    if(notice_name MATCHES "^\\.\\.")
      get_filename_component(notice_name "${notice}" NAME)
    endif()
    string(APPEND ORTOOLS_NOTICES "\n${notice_name}\n${notice_text}\n")
  endif()
endforeach()
file(GLOB_RECURSE DUCKDB_THIRD_PARTY_FILES "${CMAKE_SOURCE_DIR}/third_party/*")
foreach(notice IN LISTS DUCKDB_THIRD_PARTY_FILES)
  get_filename_component(notice_name "${notice}" NAME)
  string(TOLOWER "${notice_name}" notice_name_lower)
  if(notice_name_lower MATCHES "^(license|copying|notice|copyright)")
    file(READ "${notice}" notice_text)
    file(RELATIVE_PATH notice_name "${CMAKE_SOURCE_DIR}" "${notice}")
    string(APPEND ORTOOLS_NOTICES "\nduckdb/${notice_name}\n${notice_text}\n")
  endif()
endforeach()
configure_file(cmake/third_party_notices.hpp.in third_party_notices.hpp @ONLY)
include_directories(src/include)
set(ORTOOLS_SOURCES src/ortools_extension.cpp src/solver.cpp)
build_static_extension(ortools ${ORTOOLS_SOURCES})
build_loadable_extension(ortools "" ${ORTOOLS_SOURCES})
foreach(target ortools_extension ortools_loadable_extension)
  target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
  target_link_libraries(${target} ${ORTOOLS_LINK_TARGET})
  if(DUCKDB_VERSION_NUMBER VERSION_GREATER_EQUAL "2.0.0")
    target_compile_definitions(${target} PRIVATE ORTOOLS_DUCKDB_2=1)
  endif()
  if(DUCKDB_VERSION_NUMBER VERSION_LESS "1.5.0")
    target_compile_definitions(${target} PRIVATE ORTOOLS_DUCKDB_1_4=1)
  endif()
endforeach()
if(ORTOOLS_DUCKDB_STATIC_DIR)
  target_link_libraries(ortools_loadable_extension "${ORTOOLS_DUCKDB_STATIC_DIR}/libduckdb_static.a")
  if(DUCKDB_VERSION_NUMBER VERSION_LESS "1.5.0")
    file(GLOB dependencies "${ORTOOLS_DUCKDB_STATIC_DIR}/*.a")
    list(REMOVE_ITEM dependencies "${ORTOOLS_DUCKDB_STATIC_DIR}/libduckdb_static.a")
    target_link_libraries(ortools_loadable_extension ${dependencies})
  elseif(DUCKDB_VERSION_NUMBER VERSION_LESS "2.0.0")
    target_link_libraries(ortools_loadable_extension dummy_static_extension_loader)
  endif()
endif()
if(APPLE)
  set_target_properties(ortools_loadable_extension PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "@loader_path/lib")
elseif(UNIX)
  set_target_properties(ortools_loadable_extension PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN/lib" CXX_VISIBILITY_PRESET hidden)
  target_link_options(ortools_loadable_extension PRIVATE -Wl,--gc-sections -Wl,--exclude-libs,ALL)
endif()
install(TARGETS ortools_extension EXPORT "${DUCKDB_EXPORT_SET}" LIBRARY DESTINATION "${INSTALL_LIB_DIR}" ARCHIVE DESTINATION "${INSTALL_LIB_DIR}")
